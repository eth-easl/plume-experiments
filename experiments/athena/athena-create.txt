CREATE EXTERNAL TABLE customer (
    c_custkey     BIGINT,
    c_name        STRING,
    c_address     STRING,
    c_nationkey   BIGINT,
    c_phone       STRING,
    c_acctbal     DECIMAL(15,2),
    c_mktsegment  STRING,
    c_comment     STRING 
)
STORED AS PARQUET
LOCATION 's3://DATASET/customer';
